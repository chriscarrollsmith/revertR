Comprehensive asset screening and forecasting worksheet

To screen assets based on the last available prices, simply render this worksheet.

library(tidyverse)
library(tidyquant)
library(revertR)
options(scipen = 999)

S&P 500

# Fetch and prep S&P 500 data
tickers <- tq_index("SP500",use_fallback=TRUE) %>% filter(symbol != "-") %>% pull(symbol)
df <- prep_data_for_forecasting(tickers, include_today=FALSE)

# Forecast returns
data(sp500_ds_plotdata, package="revertR")
sp500_forecasts <- analyze_data(df, ds_plotdata, "moving_average_z_score_stratum", "diff_z_score_stratum")

#Classify and filter double strat results
classify_investment_thesis(sp500_forecasts) %>%
  group_by(type) %>%
  arrange(desc(average_expected_return)) %>%
  filter(row_number()<6) %>%
  select(symbol,average_expected_return,type) %>%
  print()
